因为是RGB，所以值域的差值要每通道都算，然后使用L1距离。
彩色版本使用L1距离开方，d范围到765，但是超过 3.11σ_r 之后，权重不是「很小」，而是就是 0。​ 代进去：
灰度 σ_r=12 → d ≥ 38 起全 0，表其实只要 39 项
彩色 L1 σ_r≈30 → d ≥ 94 起全 0，表只要 95 项
所以伪代码需要存在一个截断点CUT.

## 伪代码如下
# ============ 第一步：生成值域核表（离线做，σ_r 按噪声档位定）============
# 彩色版和灰度版只差三样：d 的上界、σ_r 的数值、表的长度。公式完全一样。
#   d 用 L1 颜色距离 |ΔR|+|ΔG|+|ΔB| -> 上界 765（灰度是 255，L2 欧氏是 442）
#   σ_r 必须按 L1 的量纲重新标定：≈ 20.8 × 1.4 ~ 1.5 ≈ 30（不能沿用灰度的 12）
#   63 是 6bit 无符号的满量程 = 2^6 - 1（权重 1.0 映射到最大格子）
# round指的是四舍五入到最近的整数
for d in 0 .. 765:（L1距离最大765，L2距离最大441）
    LUT_w[d] = round( 63 * exp( -(d*d) / (2*σ_r*σ_r) ) )   # 6bit 权重，0~63
D_MAX = 765
CUT   = 94                                     # 截断点 = ceil(3.11 × σ_r)，σ_r=30 时是 94
for d in 0 .. CUT:                             # ← 只算到截断点，不用算满 765 项
    LUT_w[d] = round( 63 * exp( -(d*d) / (2*σ_r*σ_r) ) )
# 为什么能截断：d > 3.11σ_r 时 63*exp(...) < 0.5，四舍五入就是 0 ——
# 后面那一段不是"很小"而是"就是 0"，不存进表零精度损失。
# 所以表只要 95 项（取 2 的幂 = 128 项）；不截断的话名义上是 766 项，88% 是浪费。

    
# ---- 运行时：彩色 3×3 窗口，逐像素 ----
function bilateral_rgb(win[3][3][3]):          # 每格是一组 (R,G,B)
    cR, cG, cB = win[1][1]
    numR = numG = numB = 0
    den  = 0                                    # 只开一个 den
    for i in 0..2, j in 0..2:
        vR, vG, vB = win[i][j]
        # ① 一次算颜色距离：三个通道的差一起用
        d = abs(vR - cR) + abs(vG - cG) + abs(vB - cB)   # L1，0~765，免开方
        # ② 一次查表得权重；超过截断点直接给 0（表里本来也没存那一段）
        wr = (d > CUT) ? 0 : LUT_w[d]           # 比较器 + mux，一个 mux 的事
        w  = SPATIAL[i][j] * wr                 # 空间核 1/2/1 是常数，0 乘法器
        # ③ 三个通道各攒一个 num，但共用同一个 den
        numR += w * vR
        numG += w * vG
        numB += w * vB
        den  += w
    # ④ 归一化：倒数算一次，三个通道各乘一次
    inv = recip(den)                             # 1/den 查表，避免三次除法
    return (round(numR * inv), round(numG * inv), round(numB * inv))


PGA实现要点
核心是这两行如何FPGA化：
 inv = recip(den)    ;                       
 return (round(numR * inv), round(numG * inv), round(numB * inv)) ;

这两行的FPAG化具体如下

// ---- 1) 倒数 ROM：内容 = round(2^20 / (addr + 252)) ----
reg [12:0] inv_rom [0:1023];
initial $readmemh("inv_rom.coe", inv_rom);

// ---- 2) 地址 + 同步读（1 拍）----
reg [9:0]  addr_q;
reg [12:0] inv_q;
reg [17:0] numR_q, numG_q, numB_q;

always @(posedge clk) begin
    addr_q <= den - 10'd252;      // 常数减法，可用 DSP 的 pre-adder
    inv_q  <= inv_rom[addr_q];    // BRAM 同步读
    numR_q <= numR;               // ★ num 必须同时打拍！
    numG_q <= numG;
    numB_q <= numB;
end

// ---- 3) 乘 + 舍入 + 移位 + 饱和（1 拍）----
//  18bit × 13bit = 30bit；+2^19 走 DSP 的 C 端口
wire [29:0] pR = numR_q * inv_q + 30'h80000;   // 0x80000 = 2^19
wire [29:0] pG = numG_q * inv_q + 30'h80000;
wire [29:0] pB = numB_q * inv_q + 30'h80000;

wire [9:0] rR = pR[29:20];                      // >> 20
wire [7:0] oR = (rR > 10'd255) ? 8'd255 : rR[7:0];   // 饱和兜底
// oG / oB 同理

always @(posedge clk) m_tdata <= {oR, oG, oB};
