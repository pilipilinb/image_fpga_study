//MHC版伪代码
/ W[5][5] = 5×5 窗口寄存器阵列（模块二输出）；W[2][2] = 中心
// pos = 相位检测（模块三）∈ {R, Gr, Gb, B}

case (pos)
  POS_R: begin                                    // 中心 R：十字 4 格是 G，对角 4 格是 B，远端十字 4 格是 R
    r_out = W[2][2];                                       // 直通（打拍对齐）
    g_out = (4*W[2][2]                                     // 旗舰核：底色(十字G) + ½(Rc − 远R)
           + 2*(W[1][2]+W[3][2]+W[2][1]+W[2][3])
           -   (W[0][2]+W[4][2]+W[2][0]+W[2][4])) >> 3;
    b_out = (4*W[2][2]                                     // 同配方：底色换成对角 4 个 B
           + 2*(W[1][1]+W[1][3]+W[3][1]+W[3][3])
           -   (W[0][2]+W[4][2]+W[2][0]+W[2][4])) >> 3;  // 细节项与 g_out 完全相同
  end
  POS_B: begin                                    // 中心 B：与 POS_R 同构，R↔B 对调
    b_out = W[2][2];                                       // 直通
    g_out = (4*W[2][2]
           + 2*(W[1][2]+W[3][2]+W[2][1]+W[2][3])            // 底色：十字 4 个 G
           -   (W[0][2]+W[4][2]+W[2][0]+W[2][4])) >> 3;  // 细节：Bc − 远端 B
    r_out = (4*W[2][2]
           + 2*(W[1][1]+W[1][3]+W[3][1]+W[3][3])            // 底色：对角 4 个 R
           -   (W[0][2]+W[4][2]+W[2][0]+W[2][4])) >> 3;
  end
  POS_GR: begin                                   // 中心 Gr：R 在左右、B 在上下，远端 G 跟邻居同方向
    g_out = W[2][2];                                       // 直通
    r_out = (2*W[2][2]                                     // G 位版：底色(左右R) + ½(Gc − 左右远G)
           + 2*(W[2][1]+W[2][3])
           -   (W[2][0]+W[2][4])) >> 2;
    b_out = (2*W[2][2]                                     // 同配方竖过来：底色(上下B) + ½(Gc − 上下远G)
           + 2*(W[1][2]+W[3][2])
           -   (W[0][2]+W[4][2])) >> 2;
  end
  POS_GB: begin                                   // 中心 Gb：B 在左右、R 在上下（Gr 的 R↔B 对调）
    g_out = W[2][2];                                       // 直通
    b_out = (2*W[2][2]
           + 2*(W[2][1]+W[2][3])                            // 底色：左右 2 个 B
           -   (W[2][0]+W[2][4])) >> 2;
    r_out = (2*W[2][2]
           + 2*(W[1][2]+W[3][2])                            // 底色：上下 2 个 R
           -   (W[0][2]+W[4][2])) >> 2;
  end
endcase

//双线性插值版伪代码
// W[5][5] = 5×5 窗口寄存器阵列（模块二输出）；W[2][2] = 中心
// pos = 相位检测（模块三）∈ {R, Gr, Gb, B}

case (pos)
  POS_R: begin                              // 中心 R：十字 4 格是 G，对角 4 格是 B
    r_out = W[2][2];                       // 直通（打拍对齐）
    g_out = (W[1][2]+W[3][2]+W[2][1]+W[2][3]) >> 2;   // 4 个 G 平均
    b_out = (W[1][1]+W[1][3]+W[3][1]+W[3][3]) >> 2;   // 4 个 B 平均
  end
  POS_B: begin                              // 中心 B：与 POS_R 同构，R↔B 对调
    b_out = W[2][2];                       // 直通
    g_out = (W[1][2]+W[3][2]+W[2][1]+W[2][3]) >> 2;   // 4 个 G 平均
    r_out = (W[1][1]+W[1][3]+W[3][1]+W[3][3]) >> 2;   // 4 个 R 平均
  end
  POS_GR: begin                             // 中心 Gr：R 在左右、B 在上下
    g_out = W[2][2];                       // 直通
    r_out = (W[2][1]+W[2][3]) >> 1;         // 左右 2 个 R 平均
    b_out = (W[1][2]+W[3][2]) >> 1;         // 上下 2 个 B 平均
  end
  POS_GB: begin                             // 中心 Gb：B 在左右、R 在上下
    g_out = W[2][2];                       // 直通
    b_out = (W[2][1]+W[2][3]) >> 1;         // 左右 2 个 B 平均
    r_out = (W[1][2]+W[3][2]) >> 1;         // 上下 2 个 R 平均
  end
endcase
